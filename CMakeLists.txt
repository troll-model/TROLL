cmake_minimum_required(VERSION 3.13)

project(TROLL VERSION 4.1.5 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
message(STATUS "CXX compiler ${CMAKE_CXX_COMPILER}, build type ${CMAKE_BUILD_TYPE}")

# --- Options -----------------------------------------------------------------
option(TROLL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors (-Werror); ON in CI" OFF)
option(TROLL_EXTRA_WARNINGS "Add -Wextra -Wshadow -Wconversion (advisory: several hundred warnings today)" OFF)
option(TROLL_OPENMP "Parallelise leaf fluxes and the LAImax table with OpenMP (results do not depend on the thread count)" ON)
option(TROLL_LTO "Link-time optimisation (results unchanged, a few percent faster)" ON)
set(TROLL_SANITIZE "" CACHE STRING "Semicolon-separated sanitizers, e.g. address;undefined")
set(TROLL_DEFINES "" CACHE STRING
    "Compile-time model options, e.g. \"Output_ABC;TROLL_NO_WATER\" (see include/troll_defines.hpp)")

# --- Dependencies --------------------------------------------------------------
find_package(GSL REQUIRED)
message(STATUS "GSL ${GSL_VERSION} found in ${GSL_INCLUDE_DIRS}")

# --- Executable ----------------------------------------------------------------
add_executable(${PROJECT_NAME}
    src/troll.cpp
    src/species.cpp
    src/crown.cpp
    src/tree.cpp
    src/lookup_tables.cpp
    src/input.cpp
    src/memory.cpp
    src/output.cpp
    src/simulation.cpp
    src/internal/lookup_tables_cache.cpp
    src/internal/param_registry.cpp
)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/internal
)
target_link_libraries(${PROJECT_NAME} PRIVATE GSL::gsl GSL::gslcblas)
target_compile_definitions(${PROJECT_NAME} PRIVATE TROLL_VERSION="${PROJECT_VERSION}")

if(TROLL_OPENMP)
    find_package(OpenMP)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(${PROJECT_NAME} PRIVATE OpenMP::OpenMP_CXX)
        message(STATUS "OpenMP enabled (set OMP_NUM_THREADS to choose the number of threads)")
    else()
        message(WARNING "OpenMP not found: building single-threaded. On macOS: brew install libomp")
    endif()
endif()

if(TROLL_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT _troll_ipo OUTPUT _troll_ipo_msg)
    if(_troll_ipo AND NOT TROLL_SANITIZE)
        set_property(TARGET ${PROJECT_NAME} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
        message(STATUS "Link-time optimisation enabled")
    endif()
endif()

# Warnings are ON (the former global "-w" silently disabled all of them).
target_compile_options(${PROJECT_NAME} PRIVATE -Wall)
if(TROLL_EXTRA_WARNINGS)
    target_compile_options(${PROJECT_NAME} PRIVATE -Wextra -Wshadow -Wconversion)
endif()
if(TROLL_WARNINGS_AS_ERRORS)
    target_compile_options(${PROJECT_NAME} PRIVATE -Werror)
endif()

if(TROLL_DEFINES)
    message(STATUS "TROLL model options: ${TROLL_DEFINES}")
    target_compile_definitions(${PROJECT_NAME} PRIVATE ${TROLL_DEFINES})
endif()

if(TROLL_SANITIZE)
    string(REPLACE ";" "," _troll_sanitizers "${TROLL_SANITIZE}")
    message(STATUS "Sanitizers: ${_troll_sanitizers}")
    target_compile_options(${PROJECT_NAME} PRIVATE -fsanitize=${_troll_sanitizers} -fno-omit-frame-pointer -g)
    target_link_options(${PROJECT_NAME} PRIVATE -fsanitize=${_troll_sanitizers})
endif()

# --- Regression checks (see sanity/README.md) --------------------------------------
set(_troll_check ${CMAKE_SOURCE_DIR}/sanity/check.sh --troll $<TARGET_FILE:${PROJECT_NAME}>)

# Since 4.1.2 the regression checks compare against reference outputs stored in
# sanity/golden/ (outputs < 1 MB of each case, generated by `make golden`). 4.1.2 changed
# results on purpose (Photosyn CIJ fix, accurate electron transport), so the upstream 4.0
# reference no longer applies; check.sh --ref <commit> still builds and runs any commit.
#
# Tolerances: on the platform that generated the references (Linux x86-64, GCC 13, glibc)
# results are bit-identical, and tolerances can be 0 for refactoring work
#   cmake -DTROLL_SANITY_RTOL_EXAMPLE=0 -DTROLL_SANITY_RTOL_LONG=0 -DTROLL_SANITY_ATOL=0 \
#         -DTROLL_SANITY_FILES_LONG=. ..
# The defaults allow for other compilers/libm: the lookup tables are computed with exp/log,
# whose last bits differ between math libraries; after a few months such a difference tips
# a mortality draw and per-tree outputs diverge, so long runs compare stand-level aggregates.
set(TROLL_SANITY_RTOL_EXAMPLE "1e-2" CACHE STRING "Relative tolerance, 20-day example")
set(TROLL_SANITY_RTOL_LONG "1e-2" CACHE STRING "Relative tolerance, long stand runs (aggregates)")
set(TROLL_SANITY_ATOL "1e-6" CACHE STRING "Absolute tolerance (ignores negligible values)")
set(TROLL_SANITY_FILES_LONG "sumstats|water_balance|LAIprofile" CACHE STRING "Output files compared for long runs (regex)")
set(_troll_tol_example --rtol ${TROLL_SANITY_RTOL_EXAMPLE} --atol ${TROLL_SANITY_ATOL})
set(_troll_tol_long --rtol ${TROLL_SANITY_RTOL_LONG} --atol ${TROLL_SANITY_ATOL} --files ${TROLL_SANITY_FILES_LONG})

# 20 days from bare ground on the 4 ha example (fast smoke test)
add_custom_target(sanity
    COMMAND ${_troll_check} --case example --nbiter 20 --golden golden/example_n20 ${_troll_tol_example}
    DEPENDS ${PROJECT_NAME} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/sanity
    COMMENT "Sanity check: example, 20 days" USES_TERMINAL VERBATIM)

# One year on an established 1 ha stand: adult growth, competition, mortality, treefall
add_custom_target(sanity-stand
    COMMAND ${_troll_check} --case stand --nbiter 365 --golden golden/stand_n365 ${_troll_tol_long}
    DEPENDS ${PROJECT_NAME} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/sanity
    COMMENT "Sanity check: established stand, 1 year" USES_TERMINAL VERBATIM)

# 1000 days on the established stand (long-run dynamics; ~4 min single-threaded)
add_custom_target(sanity1000
    COMMAND ${_troll_check} --case stand --nbiter 1000 --golden golden/stand_n1000 ${_troll_tol_long}
    DEPENDS ${PROJECT_NAME} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/sanity
    COMMENT "Sanity check: established stand, 1000 days" USES_TERMINAL VERBATIM)

# Regenerate the stored reference outputs. Only when results are MEANT to change:
# review the differences first, then commit the new references on their own.
add_custom_target(golden
    COMMAND ${_troll_check} --case example --nbiter 20 --update-golden golden/example_n20
    COMMAND ${_troll_check} --case stand --nbiter 365 --update-golden golden/stand_n365
    COMMAND ${_troll_check} --case stand --nbiter 1000 --update-golden golden/stand_n1000
    DEPENDS ${PROJECT_NAME} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/sanity
    COMMENT "Regenerating stored reference outputs in sanity/golden/" USES_TERMINAL VERBATIM)

# Unit tests of kernel functions (tests/)
add_executable(test_quad EXCLUDE_FROM_ALL tests/test_quad.cpp)
target_include_directories(test_quad PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(test_quad PRIVATE -Wall)
if(TROLL_WARNINGS_AS_ERRORS)
    target_compile_options(test_quad PRIVATE -Werror)
endif()
add_custom_target(unit-tests
    COMMAND test_quad
    DEPENDS test_quad
    COMMENT "Unit tests: quadratic solvers of the leaf kernel" USES_TERMINAL VERBATIM)

add_custom_target(check DEPENDS unit-tests sanity sanity-stand)

# Outputs must be bit-identical whatever the number of OpenMP threads
add_custom_target(threads-check
    COMMAND ${CMAKE_SOURCE_DIR}/sanity/bench.sh --troll $<TARGET_FILE:${PROJECT_NAME}> --threads "1 4" --nbiter 30
    DEPENDS ${PROJECT_NAME} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/sanity
    COMMENT "Thread-count independence: 1 vs 4 threads must give identical outputs" USES_TERMINAL VERBATIM)

# --- API documentation (Doxygen) ------------------------------------------------
# make docs        HTML documentation in <build>/docs/html (open index.html)
# make docs-check  regenerate it, report coverage, fail on documentation errors
find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    set(TROLL_DOCS_DIR ${CMAKE_BINARY_DIR}/docs)
    configure_file(${CMAKE_SOURCE_DIR}/docs/Doxyfile.in ${CMAKE_BINARY_DIR}/Doxyfile @ONLY)
    add_custom_target(docs
        COMMAND ${CMAKE_COMMAND} -E remove_directory ${TROLL_DOCS_DIR}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${TROLL_DOCS_DIR}
        COMMAND ${DOXYGEN_EXECUTABLE} ${CMAKE_BINARY_DIR}/Doxyfile
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
        COMMENT "Generating API documentation in ${TROLL_DOCS_DIR}/html" VERBATIM)
    if(Python3_Interpreter_FOUND)
        add_custom_target(docs-check
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/docs/doc_coverage.py ${TROLL_DOCS_DIR}
            DEPENDS docs
            COMMENT "Documentation coverage and error check" USES_TERMINAL VERBATIM)
    endif()
else()
    message(STATUS "Doxygen not found: 'make docs' and 'make docs-check' unavailable")
endif()
